Reprodução da medição, a partir da raiz desta worktree (Node, leitura somente):

```sh
node --input-type=module <<'NODE'
import fs from 'node:fs';
const doc = fs.readFileSync('docs/maps/LAJES-VISUAL-REFERENCIAS.md', 'utf8');
const names = [...doc.matchAll(/^\| `(props|ambient)\/([^`]+\.glb)` \|/gm)]
  .map(m => `public/models/${m[1]}/${m[2]}`);
for (const file of names) {
  const b = fs.readFileSync(file);
  if (b.toString('ascii', 0, 4) !== 'glTF') throw Error(file);
  let g;
  for (let off = 12; off < b.length;) {
    const len = b.readUInt32LE(off);
    if (b.readUInt32LE(off + 4) === 0x4e4f534a)
      g = JSON.parse(b.subarray(off + 8, off + 8 + len).toString());
    off += len + 8;
  }
  let triangles = 0;
  for (const m of g.meshes) for (const p of m.primitives) {
    if ((p.mode ?? 4) !== 4) throw Error(`Non-triangle primitive: ${file}`);
    triangles += g.accessors[p.indices ?? p.attributes.POSITION].count / 3;
  }
  console.log(file, b.length, triangles, g.animations?.length ?? 0);
}
NODE
```
